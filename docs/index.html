Quick Start

# Install directly
curl -sSL https://raw.githubusercontent.com/jiab77/ai-pipeline/main/install.sh | bash


# And run it
aide help


# Or clone and enter
git clone --recursive https://github.com/jiab77/ai-pipeline.git && cd ai-pipeline


# Start chatting with default provider
./cli.sh


# Start chatting with another provider
./cli.sh provider openrouter


# Start chatting with another provider + fallback provider
./cli.sh provider openrouter fallback groq


# Or go straight to the web portal
./cli.sh server web
